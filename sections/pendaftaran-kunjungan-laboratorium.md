\section{Pendaftaran Kunjungan Laboratorium}

\par Modul ini menjelaskan cara mendaftarkan pasien ke Laboratorium yang dilakukan oleh staf admisi rumah sakit. Pendaftaran kunjungan laboratorium dapat dilakukan untuk pasien yang sudah terdaftar maupun pasien baru.

Langkah-langkah melakukan pendaftaran lab:

\begin{itemize}
    \item Buka menu \textbf{Kunjungan Pasien} kemudian pilih \textbf{Kunjungan Rawat Jalan}. Klik tombol \textbf{Buat Kunjungan RJ} yang berada di pojok kanan atas halaman.
        \begin{figure}[H]
        \centering
            \includegraphics[width=\textwidth,page=1]{images/pdf/pendaftaran-kunjungan-laboratorium.pdf}
        \caption{Halaman Kunjungan Rawat Jalan}
        \label{fig:pendaftaran-kunjungan-laboratorium-1}
        \end{figure}

    \item Setelah muncul modal pilihan jenis kunjungan, pilih \textbf{LAB} untuk membuat kunjungan laboratorium.
        \begin{figure}[H]
        \centering
            \includegraphics[width=\textwidth,page=2]{images/pdf/pendaftaran-kunjungan-laboratorium.pdf}
        \caption{Modal pilihan jenis kunjungan}
        \label{fig:pendaftaran-kunjungan-laboratorium-2}
        \end{figure}

    \item Sistem akan menampilkan halaman \textbf{Kunjungan Laboratorium}. Halaman ini memiliki dua tab utama: \textbf{Rujukan Lab} dan \textbf{Kunjungan Lab}.
        \begin{figure}[H]
        \centering
            \includegraphics[width=\textwidth,page=3]{images/pdf/pendaftaran-kunjungan-laboratorium.pdf}
        \caption{Halaman Kunjungan Laboratorium - Tab Rujukan Lab}
        \label{fig:pendaftaran-kunjungan-laboratorium-3}
        \end{figure}

    \item Pilih tab \textbf{Kunjungan Lab} untuk melihat daftar kunjungan laboratorium. Klik tombol \textbf{+ Kunjungan Laboratorium} untuk mendaftarkan kunjungan baru.
        \begin{figure}[H]
        \centering
            \includegraphics[width=\textwidth,page=4]{images/pdf/pendaftaran-kunjungan-laboratorium.pdf}
        \caption{Tab Kunjungan Lab dengan tombol tambah kunjungan}
        \label{fig:pendaftaran-kunjungan-laboratorium-4}
        \end{figure}

    \item Form \textbf{Pendaftaran Pasien Laboratorium Baru} akan muncul. Form ini terdiri dari beberapa bagian: Data Pasien, Pendaftaran Pasien Laboratorium Baru, Penjamin, dan Layanan Laboratorium. Untuk mencari pasien yang sudah terdaftar, klik tombol \textbf{Cari No Rekam Medis}. Untuk mendaftarkan pasien baru, klik tombol \textbf{Pasien Baru}.
        \begin{figure}[H]
        \centering
            \includegraphics[width=\textwidth,page=5]{images/pdf/pendaftaran-kunjungan-laboratorium.pdf}
        \caption{Form Pendaftaran Pasien Laboratorium Baru}
        \label{fig:pendaftaran-kunjungan-laboratorium-5}
        \end{figure}

    \item Jika memilih \textbf{Cari No Rekam Medis}, akan muncul modal pencarian pasien. Cari pasien berdasarkan nama, nomor rekam medis, atau data lainnya. Klik pada baris pasien yang diinginkan untuk memilih.
        \begin{figure}[H]
        \centering
            \includegraphics[width=\textwidth,page=6]{images/pdf/pendaftaran-kunjungan-laboratorium.pdf}
        \caption{Modal pencarian pasien}
        \label{fig:pendaftaran-kunjungan-laboratorium-6}
        \end{figure}

    \item Setelah memilih pasien, data pasien akan terisi secara otomatis pada form. Selanjutnya, pilih penjamin dengan mengklik tombol $\vcenter{\hbox{\includegraphics[height=.6cm, page=9]{images/pdf/his-v3-baru-2025.pdf}}}$ pada bagian Penjamin.
        \begin{figure}[H]
        \centering
            \includegraphics[width=\textwidth,page=7]{images/pdf/pendaftaran-kunjungan-laboratorium.pdf}
        \caption{Form dengan data pasien yang sudah terisi}
        \label{fig:pendaftaran-kunjungan-laboratorium-7}
        \end{figure}

    \item Lengkapi informasi rujukan jika ada, seperti \textbf{Dokter/RS Perujuk}, \textbf{Tindakan Rujukan}, dan \textbf{Dokter Pengirim}. Untuk menambahkan layanan pemeriksaan laboratorium, klik tombol \textbf{+ Pemeriksaan} pada bagian Layanan Laboratorium.
        \begin{figure}[H]
        \centering
            \includegraphics[width=\textwidth,page=8]{images/pdf/pendaftaran-kunjungan-laboratorium.pdf}
        \caption{Form dengan data rujukan dan bagian Layanan Laboratorium}
        \label{fig:pendaftaran-kunjungan-laboratorium-8}
        \end{figure}

    \item Modal pemilihan layanan laboratorium akan muncul. Layanan dikelompokkan berdasarkan kategori seperti \textbf{H. Alergi}, \textbf{Q. Analisa Lain-Lain}, \textbf{D. Antibodi}, \textbf{N. Biomonitoring}, dan lainnya. Centang layanan yang diinginkan, kemudian klik tombol \textbf{Simpan}.
        \begin{figure}[H]
        \centering
            \includegraphics[width=\textwidth,page=10]{images/pdf/pendaftaran-kunjungan-laboratorium.pdf}
        \caption{Modal pemilihan layanan laboratorium}
        \label{fig:pendaftaran-kunjungan-laboratorium-10}
        \end{figure}

    \item Setelah semua data terisi lengkap, klik tombol \textbf{Simpan} untuk menyimpan pendaftaran kunjungan laboratorium.

    \item Pendaftaran berhasil. Data kunjungan laboratorium akan muncul pada daftar di tab \textbf{Kunjungan Lab}.
        \begin{figure}[H]
        \centering
            \includegraphics[width=\textwidth,page=11]{images/pdf/pendaftaran-kunjungan-laboratorium.pdf}
        \caption{Daftar kunjungan laboratorium yang sudah terdaftar}
        \label{fig:pendaftaran-kunjungan-laboratorium-11}
        \end{figure}
\end{itemize}

\begin{mdframed}[backgroundcolor=yellow!20]
\textbf{Catatan:} Pastikan data penjamin dan layanan laboratorium sudah dipilih dengan benar sebelum menyimpan pendaftaran.
\end{mdframed}
